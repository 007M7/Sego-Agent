% fig1-motivation.tex — Figure 1: three risk chains (motivation)
\documentclass[tikz,border=10pt]{standalone}
\usetikzlibrary{arrows.meta,positioning,fit,backgrounds,calc}

\begin{document}
\begin{tikzpicture}[
  font=\small,
  box/.style={draw, rounded corners=2pt, thick, minimum height=9mm, align=center, inner sep=3pt},
  bad/.style={fill=red!8, draw=red!45!black},
  mid/.style={fill=orange!10, draw=orange!55!black},
  good/.style={fill=green!10, draw=green!45!black},
  chain/.style={-{Stealth[length=2.2mm]}, thick, draw=red!50!black},
  sarr/.style={-{Stealth[length=2.2mm]}, thick, draw=green!45!black},
  panel/.style={font=\footnotesize\bfseries, anchor=north},
  lab/.style={font=\footnotesize, text=black!60}
]

% ================= Panel 1 =================
\node[panel, text=red!50!black] at (0, 1.1) {Risk 1 · Review co-generated with code};
\node[box, bad]   (p1a) at (0, 0)    {AI tool generates the change};
\node[box, bad]   (p1b) at (0, -1.4) {Same toolchain reviews itself};
\node[box, mid]   (p1c) at (0, -2.8) {Defects shipped};
\draw[chain] (p1a) -- (p1b);
\draw[chain] (p1b) -- node[lab, right] {conflict of interest} (p1c);

% ================= Panel 2 =================
\node[panel, text=red!50!black] at (6.2, 1.1) {Risk 2 · ``Done'' $\neq$ done};
\node[box, bad]   (p2a) at (6.2, 0)    {Model claims completion};
\node[box, mid]   (p2b) at (6.2, -1.4) {Missing evidence / unverified items};
\node[box, bad]   (p2c) at (6.2, -2.8) {Gaps silently swallowed};
\draw[chain] (p2a) -- (p2b);
\draw[chain] (p2b) -- node[lab, right] {no evidence chain} (p2c);

% ================= Panel 3 =================
\node[panel, text=red!50!black] at (12.4, 1.1) {Risk 3 · Results cannot be re-checked};
\node[box, mid]   (p3a) at (12.4, 0)    {Findings lack evidence binding};
\node[box, mid]   (p3b) at (12.4, -1.4) {Zero findings treated as a pass};
\node[box, bad]   (p3c) at (12.4, -2.8) {No audit trail / not reproducible};
\draw[chain] (p3a) -- (p3b);
\draw[chain] (p3b) -- node[lab, right] {no artifact trail} (p3c);

% ================= Sego bar =================
\node[box, good, minimum width=140mm, minimum height=10mm] (sego) at (6.2, -4.4)
  {\textbf{Sego}: explicit scope $\rightarrow$ constrained model review $+$ deterministic evidence gate $\rightarrow$ structured artifacts $\rightarrow$ human decision (zero findings $\neq$ pass)};
\foreach \x in {0, 6.2, 12.4} \draw[sarr] (\x, -3.5) -- (\x, -3.9) -- (sego.north -| \x, -3.9);

\begin{scope}[on background layer]
  \node[draw=black!12, dashed, rounded corners=3pt, fit=(p1a)(p3c), inner sep=6pt] {};
\end{scope}
\end{tikzpicture}
\end{document}
